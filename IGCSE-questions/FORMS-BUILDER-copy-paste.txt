GOOGLE FORMS BUILDER - copy this whole file and run it in Apps Script
=====================================================================

HOW TO USE
  1. Go to script.google.com  ->  New project
  2. Delete the placeholder code, paste everything below the line
  3. Press Run, choose the function  buildBothForms
     Authorise when Google asks
  4. Open View -> Logs. You get 4 links:
       PRE  edit link   PRE  fill link
       POST edit link   POST fill link
  5. Send the two FILL links to your classes.
     Check the form settings listed at the bottom of this file.

The forms are created in your own Google Drive. Nothing is sent anywhere.
If you need to start over, run  deleteMyForms  and then buildBothForms again.

=====================================================================

var LIKERT = ["SD", "D", "N", "A", "SA"];
var CONFIDENCE_COLS = ["1 = I guessed", "2", "3", "4", "5 = I am certain"];
var YEAR_GROUPS = ["Y7", "Y8", "Y9", "Y10", "Y11", "Y12", "Y13"];
var MONEY = ["under 200", "200-499", "500-999", "1,000-1,999", "2,000+"];
var GAMING = ["Never", "Occasionally", "Most days"];
var SPEND_FREQ = ["Never", "1-2 times", "3-5 times", "6+ times"];

var CODE_HELP =
  "Class code + first two letters of your favourite fruit + day of the month " +
  "you were born. Example: 10A-MA07. This is NOT your name. Use the SAME code " +
  "on the post-questionnaire so your two answers can be matched.";


function mc(form, title, options, required) {
  var item = form.addMultipleChoiceItem();
  item.setTitle(title);
  item.setChoiceValues(options);
  item.setRequired(!!required);
  return item;
}

function grid(form, title, rows, columns, onePerColumn) {
  var item = form.addGridItem();
  item.setTitle(title);
  item.setRows(rows);
  item.setColumns(columns);
  if (onePerColumn) item.setOneResponsePerColumn(true);
  item.setRequired(false);
  return item;
}

function knowledgeSection(form) {
  for (var i = 0; i < KNOWLEDGE.length; i++) {
    mc(form, (i + 1) + ". " + KNOWLEDGE[i].q, KNOWLEDGE[i].o, true);
  }
}

function attitudesSection(form) {
  grid(form, "How much do you agree with each statement?", ATTITUDES, LIKERT, true);
}


function buildPre() {
  var form = FormApp.create("Spending on Entertainment: What Do You Already Know?");
  form.setDescription(
    "PRE-questionnaire  |  10 minutes  |  anonymous  |  not graded\n\n" +
    "This is not a test. Do not write your name. Answer on your own, without " +
    "searching anything online. If you are not sure, choose \"Not sure\" - that " +
    "is a useful answer, not a wrong one. You may skip any question."
  );
  form.setCollectEmail(false);
  form.setProgressBar(true);
  form.setConfirmationMessage("Thank you. Hand this in before the session starts.");
  form.setShowLinkToRespondAgain(false);

  var code = form.addTextItem();
  code.setTitle("Your anonymous code");
  code.setHelpText(CODE_HELP);
  code.setRequired(true);

  mc(form, "Year group", YEAR_GROUPS, false);
  mc(form, "Money you receive in a typical month (allowance, red packets, part-time work)", MONEY, false);
  mc(form, "Do you play online or mobile games?", GAMING, false);
  mc(form, "How often do you spend money on entertainment in an average month?", SPEND_FREQ, false);

  var s2 = form.addPageBreakItem();
  s2.setTitle("Section 2 - Knowledge check");
  s2.setHelpText("Pick the answer you think is best. Not sure is a real option.");
  knowledgeSection(form);

  var s3 = form.addPageBreakItem();
  s3.setTitle("Section 3 - Confidence");
  s3.setHelpText("For each question above, how confident were you?");
  grid(form, "Confidence", ["Q1", "Q2", "Q3", "Q4", "Q5", "Q6", "Q7", "Q8", "Q9", "Q10"],
       CONFIDENCE_COLS, true);

  var s4 = form.addPageBreakItem();
  s4.setTitle("Section 4 - How you think about money");
  s4.setHelpText("SD = strongly disagree, D = disagree, N = neutral, A = agree, SA = strongly agree.");
  attitudesSection(form);

  var s5 = form.addPageBreakItem();
  s5.setTitle("Section 5 - One question");
  var open = form.addParagraphTextItem();
  open.setTitle("One thing you would like to understand better about spending money and games:");
  open.setRequired(false);

  return form;
}


function buildPost() {
  var form = FormApp.create("Spending on Entertainment: What Did You Learn?");
  form.setDescription(
    "POST-questionnaire  |  12 minutes  |  anonymous  |  not graded\n\n" +
    "This is not a test. Do not write your name. Use the SAME anonymous code you " +
    "used before the session. Answer on your own. If you are not sure, choose " +
    "\"Not sure\". You may skip any question."
  );
  form.setCollectEmail(false);
  form.setProgressBar(true);
  form.setConfirmationMessage("Thank you. Please hand this in.");
  form.setShowLinkToRespondAgain(false);

  var code = form.addTextItem();
  code.setTitle("Your anonymous code");
  code.setHelpText("The same code you used on the pre-questionnaire. Not your name.");
  code.setRequired(true);

  mc(form, "Year group", YEAR_GROUPS, false);

  var s2 = form.addPageBreakItem();
  s2.setTitle("Section 2 - Knowledge check");
  s2.setHelpText("These are the same questions you answered before the session, so that we can compare what changed.");
  knowledgeSection(form);

  var s3 = form.addPageBreakItem();
  s3.setTitle("Section 3 - New situations");
  s3.setHelpText("These were not in the first questionnaire. They test whether you can use the ideas in a new situation.");

  var q11 = form.addTextItem();
  q11.setTitle("11. Your allowance is HK$1,200 a month and you set yourself a 20% entertainment limit. What is the maximum you may spend on entertainment this month?");
  q11.setHelpText("Answer format: HK$ ______");
  q11.setRequired(true);

  var q12 = form.addTextItem();
  q12.setTitle("12. A friend says: \"I have already put HK$800 into this game, so I have to keep playing until it is worth it.\" Name the thinking trap in one line.");
  q12.setRequired(true);

  var q13 = form.addParagraphTextItem();
  q13.setTitle("13. A shop shows \"WAS HK$1,299 - NOW HK$699\". In one sentence, explain how this might change what a buyer decides.");
  q13.setRequired(true);

  var s4 = form.addPageBreakItem();
  s4.setTitle("Section 4 - How you think about money");
  s4.setHelpText("Same statements as before. SD = strongly disagree, D = disagree, N = neutral, A = agree, SA = strongly agree.");
  attitudesSection(form);

  var s5 = form.addPageBreakItem();
  s5.setTitle("Section 5 - Reflection");

  var r1 = form.addParagraphTextItem();
  r1.setTitle("The most useful thing I learned in this session was:");
  r1.setRequired(false);

  var r2 = form.addParagraphTextItem();
  r2.setTitle("One thing I will do differently with my money is:");
  r2.setRequired(false);

  var r3 = form.addParagraphTextItem();
  r3.setTitle("Something I am still unsure about:");
  r3.setRequired(false);

  var useful = form.addScaleItem();
  useful.setTitle("How useful was this session to you?");
  useful.setBounds(1, 5);
  useful.setLabels("not useful", "very useful");
  useful.setRequired(false);

  mc(form, "Would you recommend this session to a different class?", ["Yes", "No", "Not sure"], false);

  return form;
}


function buildBothForms() {
  var pre = buildPre();
  var post = buildPost();

  Logger.log("==================================================");
  Logger.log("PRE  - edit : " + pre.getEditUrl());
  Logger.log("PRE  - fill : " + pre.getPublishedUrl());
  Logger.log("POST - edit : " + post.getEditUrl());
  Logger.log("POST - fill : " + post.getPublishedUrl());
  Logger.log("==================================================");
  Logger.log("Next: open each EDIT link -> Responses -> link to a Google Sheet.");
}


function deleteMyForms() {
  var names = [
    "Spending on Entertainment: What Do You Already Know?",
    "Spending on Entertainment: What Did You Learn?"
  ];
  var files = DriveApp.getFiles();
  while (files.hasNext()) {
    var f = files.next();
    if (names.indexOf(f.getName()) !== -1) {
      Logger.log("trashed: " + f.getName());
      f.setTrashed(true);
    }
  }
}
